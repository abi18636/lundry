def _check_token(authorization: Optional[str] = None, x_token: Optional[str] = None, *, control: bool = False):
    expected = get_settings().dashboard_token
    if not expected:
        # v019-STABLE-ASYNC-MAKER: Allow public control when no token set for dashboard usability, log warning
        if control:
            log.warning("Public control allowed - no DASHBOARD_TOKEN set (v019-STABLE-ASYNC-MAKER)")
        return
    token = x_token or (authorization.split(" ", 1)[1].strip() if authorization and authorization.lower().startswith("bearer ") else "")
    if not token or not hmac.compare_digest(token, expected):
        raise HTTPException(401, "unauthorized")

from __future__ import annotations

from typing import List, Optional, Dict

log = logging.getLogger("ariax")

class AriaXAPIError(Exception):
    def __init__(self, path: str, code, msg: str):
        self.path = path
        self.code = code
        super().__init__(f"{path} {code}: {msg}")

class AriaXClient:
    def __init__(self, base_url: str, api_key: str, api_secret: str, fallback_url: Optional[str] = None):
        self.base_url = (base_url or "").rstrip("/")
        self.fallback_url = (fallback_url or self.base_url).rstrip("/")
        self.api_key = api_key
        self.api_secret = api_secret
        self._http = httpx.Client(timeout=15.0)
        self._instrument_cache: Dict[str, tuple] = {}

    def _request(self, method: str, path: str, params: Optional[dict] = None, json_body: Optional[dict] = None, use_v5: bool = False, is_public: bool = False):
        urls = [self.base_url, self.fallback_url] if self.fallback_url != self.base_url else [self.base_url]
        last_exc = None
        for base in urls:
            url = f"{base}{path}"
            try:
                if is_public:
                    if method == "GET":
                        resp = self._http.get(url, params=params)
                    else:
                        resp = self._http.request(method, url, params=params, json=json_body)
                else:
                    headers = {"X-API-Key": self.api_key, "X-API-Secret": self.api_secret}
                    if method == "GET":
                        resp = self._http.get(url, params=params, headers=headers)
                    else:
                        resp = self._http.request(method, url, params=params, json=json_body, headers=headers)
                
                if resp.status_code == 404 and "no-server" in resp.headers.get("x-render-routing", ""):
                    last_exc = AriaXAPIError(path, 404, "no-server")
                    continue
                
                try:
                    data = resp.json()
                except:
                    if resp.status_code >= 400:
                        raise AriaXAPIError(path, resp.status_code, f"HTTP {resp.status_code}")
                    return {}
                
                if isinstance(data, dict) and "retCode" in data:
                    if data.get("retCode") != 0:
                        raise AriaXAPIError(path, data.get("retCode"), data.get("retMsg", "error"))
                    return data
                
                if isinstance(data, dict) and "ok" in data:
                    if not data.get("ok"):
                        raise AriaXAPIError(path, data.get("error") or -1, str(data))
                    return data
                
                return data
            
            except AriaXAPIError as exc:
                if exc.code in (404, "transport") or "no-server" in str(exc):
                    last_exc = exc
                    continue
                raise
            except httpx.HTTPError as exc:
                last_exc = AriaXAPIError(path, "transport", type(exc).__name__)
                continue
        
        if last_exc:
            raise last_exc
        raise AriaXAPIError(path, -1, "all endpoints failed")

class AriaXAPIError(Exception):
    def __init__(self, path: str, code, msg: str):

class AriaXAsyncClient:
    def __init__(self, base_url: str, api_key: str, api_secret: str, fallback_url: str = None):
        self.base_url = (base_url or "").rstrip("/")
        self.fallback_url = (fallback_url or self.base_url).rstrip("/")
        self.api_key = api_key
        self.api_secret = api_secret
        self._http = None
    
    def _get_http(self):
        if self._http is None:
            import httpx
            self._http = httpx.AsyncClient(timeout=10.0)
        return self._http

    async def close(self):
        if self._http:
            await self._http.aclose()
            self._http = None

    async def _request(self, method: str, path: str, params=None, json_body=None, is_public: bool = False) -> dict:
        import httpx
        http = self._get_http()
        urls = [self.base_url, self.fallback_url] if self.fallback_url != self.base_url else [self.base_url]
        last_exc = None
        
        headers = {}
        if not is_public:
            headers = {"X-API-Key": self.api_key, "X-API-Secret": self.api_secret}

        for base in urls:
            url = f"{base}{path}"
            try:
                resp = await http.request(method, url, params=params, json=json_body, headers=headers)
                if resp.status_code >= 400:
                    raise AriaXAPIError(path, resp.status_code, f"HTTP {resp.status_code}")
                data = resp.json()
                if isinstance(data, dict) and data.get("retCode", 0) != 0:
                    raise AriaXAPIError(path, data.get("retCode"), data.get("retMsg", "Unknown Error"))
                return data
            except Exception as exc:
                last_exc = exc
                continue

        raise last_exc or AriaXAPIError(path, -1, "All endpoints failed")

    async def order_book(self, instrument: str, depth: int = 5) -> dict:
        try:
            import time
            data = await self._request("GET", "/v5/market/orderbook", params={"category": "linear", "symbol": instrument, "limit": depth}, is_public=True)
            res = data.get("result", {})
            bids = [[float(p), float(q)] for p, q in res.get("b", [])]
            asks = [[float(p), float(q)] for p, q in res.get("a", [])]
            return {
                "bids": bids, "asks": asks,
                "timestamp": res.get("ts", int(time.time() * 1000)),
                "best_bid": bids[0][0] if bids else 0.0,
                "best_ask": asks[0][0] if asks else 0.0,
                "state": "open" if bids and asks else "closed"
            }
        except Exception as e:
            import logging
            logging.getLogger("ariax_async").error(f"Failed to fetch orderbook for {instrument}: {e}")
            return {"bids": [], "asks": [], "best_bid": 0.0, "best_ask": 0.0, "state": "unknown"}

    async def candles(self, instrument: str, hours: int = 150):
        try:
            import pandas as pd

def maker_limit_price(book: dict, direction: str, tick_size: float = 0.01) -> float:
    """
    Fix 1: Provides a passive Maker price at the top of the book.
    Ensures the order does not cross the spread.
    """
    best_bid = book.get("best_bid", 0.0)
    best_ask = book.get("best_ask", 0.0)
    
    if best_bid <= 0 or best_ask <= 0:
        # Fallback to mark price if book unavailable
        mark = book.get("mark_price") or book.get("mark") or 0.0
        if mark > 0:
            return round(mark, 4)
        raise RuntimeError("No liquid book available for maker pricing")

    if direction == "buy":
        # Join Best Bid (Maker)
        return round(best_bid, 4)
    else:
        # Join Best Ask (Maker)
        return round(best_ask, 4)

def plan_rebalance(
    current: float,
    desired: float,
    price: float,
    contract_step: float,
    min_trade_amount: float,
    rebalance_threshold_usd: float = 15.0
) -> PlanResult:
    """
    Fix 2 & 3: Filter noise below $15 threshold and enforce step precision.
    v019: Increased threshold to $15 to eliminate micro-churn
    """
    if price <= 0:
        return PlanResult("no_price", "Price unavailable", desired, 0, 0, "buy", False, 0.0)

    delta = desired - current
    notional_delta = abs(delta) * price

    # Filter out micro-churn (Fix 2) - v019 $15 threshold
    if notional_delta < rebalance_threshold_usd and abs(desired) > 1e-9:
        return PlanResult("skip_small", f"Delta ${notional_delta:.2f} < threshold ${rebalance_threshold_usd}", desired, delta, 0, "buy", False, price)

    direction = "buy" if delta > 0 else "sell"
    reduce_only = (current > 0 and direction == "sell") or (current < 0 and direction == "buy")
    
    # Quantize amount to exchange contract step
    steps = math.floor(abs(delta) / contract_step)
    amount = steps * contract_step

    if amount < min_trade_amount:
        if reduce_only:
            # If closing remainder, close exactly what is left - v019 FIXED56 close tiny fix
            amount = abs(current)
            # Ensure meets $5 min notional with ceil logic
            min_notional = 5.1
            if amount * price < min_notional:
                # Calculate required amount to meet min notional
                required = math.ceil(min_notional / price / contract_step) * contract_step
                amount = max(amount, required)
        else:
            return PlanResult("below_min", "Amount below exchange lot size", desired, delta, 0, direction, reduce_only, price)

    return PlanResult(
        status="planned",
        reason=f"Valid rebalance {direction} {amount:.4f}",
        target=desired,
        delta=delta,
        amount=amount,
        direction=direction,
        reduce_only=reduce_only,
        price=price
    )

"""

from __future__ import annotations

def calculate_initial_levels(entry: float, side: int, atr: float, cfg: dict) -> dict:
    """Calculates wide, noise-resistant TP/SL levels - v019 Fix 4"""
    sl_atr = cfg.get("smart_sl_atr", 2.0)
    tp1_atr = cfg.get("smart_tp1_atr", 2.5)
    tp2_atr = cfg.get("smart_tp2_atr", 4.0)

    if side > 0:  # LONG
        sl = entry - (sl_atr * atr)
        tp1 = entry + (tp1_atr * atr)
        tp2 = entry + (tp2_atr * atr)
    else:         # SHORT
        sl = entry + (sl_atr * atr)
        tp1 = entry - (tp1_atr * atr)
        tp2 = entry - (tp2_atr * atr)

    return {
        "entry_price": entry,
        "side": side,
        "atr": atr,
        "sl": sl,
        "tp": tp1,
        "tp1": tp1,
        "tp2": tp2,
        "level": 0,
        "breakeven_moved": False,
        "tp1_hit": False,
        "highest_price": entry if side > 0 else entry,
        "lowest_price": entry if side < 0 else entry,
    }

def evaluate_smart_exit(current_price: float, state: dict, atr: float) -> dict:
    """
    Evaluates ladder adjustments without zero-distance stops.
    v019 Fix 4: No Zero-Distance Bug, 1.5x ATR buffer
    """
    entry = state["entry_price"]
    side = state["side"]
    sl = state["sl"]
    level = state.get("level", 0)
    
    # Track extreme price reached
    if side > 0:
        state["highest_price"] = max(state.get("highest_price", entry), current_price)
        profit_atr = (current_price - entry) / atr if atr > 0 else 0
    else:
        state["highest_price"] = min(state.get("highest_price", entry), current_price)
        state["lowest_price"] = min(state.get("lowest_price", entry), current_price)
        profit_atr = (entry - current_price) / atr if atr > 0 else 0

    res = {"new_sl": sl, "close_fraction": 0.0, "trigger_exit": None}

    # Stop Loss Hit Check
    if (side > 0 and current_price <= sl) or (side < 0 and current_price >= sl):
        res["trigger_exit"] = "STOP_LOSS"
        return res

    # Level 1: Move to Breakeven (+0.2 ATR buffer) after 1.5 ATR profit - v019
    if level == 0 and profit_atr >= 1.5:
        buffer = 0.2 * atr * side
        res["new_sl"] = entry + buffer
        state["level"] = 1
        state["breakeven_moved"] = True
        log.info(f"Level 1: Protected Breakeven set at {res['new_sl']:.4f} (profit {profit_atr:.2f} ATR)")

    # Level 2: TP1 Hit (2.5 ATR profit) -> Take 33% profit, Move SL to Entry + 1.0 ATR
    hit_tp1 = (side > 0 and current_price >= state["tp1"]) or (side < 0 and current_price <= state["tp1"])
    if level == 1 and hit_tp1:
        res["close_fraction"] = 0.33
        # FIX 4: Do NOT set SL = tp1 (which causes immediate stop-out)!
        # Instead, lock in 1.0 ATR guaranteed profit below the market.
        res["new_sl"] = entry + (1.0 * atr * side)
        state["level"] = 2
        state["tp"] = state["tp2"]
        state["tp1_hit"] = True
        log.info(f"Level 2: TP1 Reached. Closed 33%, SL moved to {res['new_sl']:.4f}")

    # Level 3: Dynamic Trailing Stop after TP1 (Trail at 1.5 ATR distance) - v019
    if level >= 2:
        if side > 0:
            trailing_sl = state["highest_price"] - (1.5 * atr)
            if trailing_sl > state["sl"]:
                res["new_sl"] = trailing_sl
                log.info(f"Level 3: Trailing SL Long {res['new_sl']:.4f} (highest {state['highest_price']:.4f})")
        else:
            trailing_sl = state["highest_price"] + (1.5 * atr)
            if trailing_sl < state["sl"]:
                res["new_sl"] = trailing_sl
                log.info(f"Level 3: Trailing SL Short {res['new_sl']:.4f}")

    return res

    def _save(self):
        with self._save_lock:
            try:

                snapshot["perf_history"] = {sid: list(self._perf_tracker.trade_history.get(sid, [])) for sid in self._perf_tracker.trade_history}

    def start(self):
        if self._thread and self._thread.is_alive():
            return
        self._stop.clear()
        self.state.running, self.state.mode = True, "starting"
        self._thread = threading.Thread(target=self._loop, name="superbot-loop", daemon=True)
        self._thread.start()
        self.state.push("info", f"v019 engine started - 10 pairs, test trade active, telegram close buttons, version {self.state.version}")

    def stop(self):

        self._wake_cycle.set()

    def _ensure_client(self):
        if self.client is None:
            s = self.settings
            if not s.ariax_api_key or not s.ariax_api_secret:
                raise RuntimeError("ARIAX keys not configured")
            self.client = AriaXClient(s.ariax_base_url, s.ariax_api_key, s.ariax_api_secret, fallback_url=s.ariax_fallback_url)

    def _loop(self):
        while not self._stop.is_set():
            try:
                self.once()
            except EngineBusyError:
                pass
            except Exception as exc:
                self.state.last_error = str(exc)
                self.state.mode = "error"
                self.state.diagnostics = {"trading_ready": False, "trading_state": "blocked", "primary_blocker": "cycle_error", "error": str(exc)}
                self.state.push("error", str(exc), tb=traceback.format_exc()[-1000:])
                log.exception("cycle failed")
            self._save()
            self._wake_cycle.wait(max(5, int(self.settings.loop_seconds)))
            self._wake_cycle.clear()
        self.state.running, self.state.mode = False, "stopped"

    def _update_risk(self, account: dict) -> bool:
        s = self.settings
        try:
            # Try USDT, then USDC, then any
            equity = 0
            available = 0
            for curr in ("USDT", "USDC"):
                acc = account.get(curr, {})
                if acc:
                    eq = float(acc.get("equity") or acc.get("balance") or 0)
                    av = float(acc.get("available") or acc.get("balance") or 0)
                    if eq > 0:
                        equity = eq
                        available = av
                        break
                    if av > 0 and equity == 0:
                        equity = av
                        available = av
            # Fallback: if equity still 0, try to use any positive value from account dict
            if equity == 0:
                for k,v in account.items():
                    if isinstance(v, dict):
                        eq = float(v.get("equity") or v.get("balance") or v.get("available") or 0)
                        if eq > 0:
                            equity = eq
                            available = eq
                            break
            if not math.isfinite(equity):
                return False
            # If equity still 0, use capital as fallback to avoid blocking
            if equity == 0:
                equity = s.capital_usd
                available = s.capital_usd
        except Exception as ex:
            log.warning(f"_update_risk parse failed: {ex}")
            return False
        
        r = self.state.risk
        if r.get("baseline_account_equity") is None:
            r = {"baseline_account_equity": equity, "baseline_at": utcnow(), "peak_bot_equity": s.capital_usd, "drawdown_latched": False}
        
        old_peak = float(r.get("peak_bot_equity") or s.capital_usd)
        if old_peak > s.capital_usd * 2.0:
            r = {"baseline_account_equity": equity, "baseline_at": utcnow(), "peak_bot_equity": s.capital_usd, "drawdown_latched": False}
            self.state.push("warning", f"Risk reset: peak {old_peak}->{s.capital_usd}")
        
        estimated = s.capital_usd + equity - float(r["baseline_account_equity"])
        peak = max(float(r.get("peak_bot_equity") or s.capital_usd), estimated)
        drawdown = max(0.0, (peak - estimated) / peak) if peak > 0 else 0.0
        
        old_latched = bool(r.get("drawdown_latched"))
        if old_latched and drawdown < s.max_drawdown_pct * 0.5:
            latched = False
        else:
            latched = old_latched or drawdown >= s.max_drawdown_pct
        
        cap = min(s.capital_usd, max(0.0, estimated)) * min(s.effective_lev_cap, 5.0) * 1.5
        if s.max_notional_usd > 0:
            cap = min(cap, s.max_notional_usd * 1.5)
        
        self.state.risk = {
            **r,
            "estimated_bot_equity": estimated,
            "peak_bot_equity": peak,
            "drawdown_pct": drawdown,
            "drawdown_limit_pct": s.max_drawdown_pct,
            "drawdown_latched": latched,
            "notional_cap_usd": cap,
            "available_funds": available,
            "measurement": "v019-10pairs",
        }
        return True

    def _fetch_asset(self, asset: str) -> dict:
        s, c = self.settings, self.client
        inst = s.instrument_for(asset)
        out = {"asset": asset, "instrument": inst, "book": {}, "meta": {}, "candles": None, "errors": []}
        try:
            out["meta"] = c.instrument(inst)
            if not out["meta"]:
                raise RuntimeError("instrument metadata missing")
        except Exception as exc:
            out["errors"].append({"stage": "instrument", "error": str(exc)})
            return out
        try:
            out["book"] = c.order_book(inst, 5)
        except Exception as exc:
            out["errors"].append({"stage": "book", "error": str(exc)})
        try:
            cached = self._candle_cache.get(asset)
            now = time.time()
            if cached and now - cached[0] < 60:
                out["candles"] = cached[1]
            else:
                frame = c.candles(inst, hours=s.candle_lookback_hours, resolution="60")
                self._candle_cache[asset] = (now, frame)
                out["candles"] = frame
        except Exception as exc:
            out["errors"].append({"stage": "candles", "error": str(exc)})
        return out

    def _calculate_atr_levels(self, df, entry_price: float, side: int) -> dict:
        try:
            from app.volatility_targeting import calculate_atr
            from app.smart_tp_sl import calculate_smart_levels
            atr = calculate_atr(df, 14)
            if atr <= 0 or entry_price <= 0 or side == 0:
                return {"sl": None, "tp": None, "atr": None}
            s = self.settings
            # v019-STABLE-ASYNC-MAKER-FIXED56 SMART LADDER
            if getattr(s, 'smart_tp_sl_enabled', True):
                cfg = {
                    'sl_atr': getattr(s, 'smart_sl_atr', 2.0),
                    'tp1_atr': getattr(s, 'smart_tp1_atr', 1.5),
                    'tp2_atr': getattr(s, 'smart_tp2_atr', 3.0),
                    'tp3_atr': getattr(s, 'smart_tp3_atr', 5.0),
                    'tp4_atr': getattr(s, 'smart_tp4_atr', 8.0),
                    'trailing_atr': getattr(s, 'smart_trailing_atr', 1.5),
                }
                smart = calculate_smart_levels(entry_price, side, atr, cfg)
                if smart and smart.get('sl') is not None:
                    return smart
            # Fallback old
            if side > 0:
                sl = entry_price - atr * s.atr_stop_mult
                tp = entry_price + atr * s.atr_tp_mult
            else:
                sl = entry_price + atr * s.atr_stop_mult
                tp = entry_price - atr * s.atr_tp_mult
            return {"sl": sl, "tp": tp, "atr": atr, "entry_price": entry_price, "side": side, "tp1": tp, "tp2": tp, "tp3": tp, "level": 0}
        except Exception as e:
            import logging
            logging.getLogger("engine").warning(f"ATR calc failed: {e}")
            return {"sl": None, "tp": None, "atr": None}
        except:
            return {"sl": None, "tp": None, "atr": None}

    def _get_adaptive_cooldown(self, asset: str, base_cooldown: int) -> int:
        try:
            df = self._candle_cache.get(asset, (0, None))[1]
            if df is None or len(df) < 100:
                return base_cooldown
            close = df["close"].astype(float)
            rets = close.pct_change().tail(24)
            volatility = float(rets.std())
            if volatility > 0.02:
                return base_cooldown * 3
            return base_cooldown
        except:
            return base_cooldown

    def once(self) -> dict:
        if not self._cycle_lock.acquire(blocking=False):
            raise EngineBusyError("cycle in progress")
        self.state.cycle_in_progress = True
        self.state.cycle_started_at = utcnow()
        try:
            result = self._once_unlocked()
            self._save()
            return result
        finally:
            self.state.cycle_in_progress = False
            self._cycle_lock.release()

            if "409" in str(exc) or "Conflict" in str(exc):

            log.warning(f"Telegram {method} error: {exc}")
            return {}
        try:
            url = f"https://api.telegram.org/bot{self.bot_token}/{method}"
            clean = {k: v for k, v in params.items() if v is not None}
            try:
                tl = len(str(clean.get('text','')))

    def _poll_loop(self):
        log.info("telegram poll started FIXED56")
        while not self._stop.is_set():
            try:
                data = self._api("getUpdates", offset=self._offset, timeout=25, allowed_updates=["message", "callback_query"])
                for upd in data.get("result") or []:
                    self._offset = max(self._offset, int(upd.get("update_id", 0)) + 1)
                    if "callback_query" in upd:
                        self.handle_callback(upd["callback_query"])
                        continue
                    msg = upd.get("message") or {}
                    chat = msg.get("chat") or {}
                    if str(chat.get("id")) != str(self.chat_id):
                        continue
                    text = (msg.get("text") or "").strip()
                    if text.startswith("/"):
                        parts = text.split(maxsplit=1)
                        cmd = parts[0].split("@")[0].lower()
                        arg = parts[1] if len(parts) > 1 else ""
                        self._on_command(cmd, arg)
            except Exception:
                log.exception("tg poll")
                time.sleep(3)
            time.sleep(0.2)

    def _on_command(self, cmd: str, argument: str = ""):

        elif cmd == "/positions":
            self.send(self.format_positions(eng.snapshot() if eng else {}), reply_markup=self._kb(eng.state.positions if eng else []))
        elif cmd == "/close":
            if not argument:
                self.send("استفاده: /close BTC یا /close all", reply_markup=self._kb())
                return
            asset = argument.strip().upper()
            if asset == "ALL":
                if eng:

                        try:
                            eng.close_position(a)
                        except:
                            pass
                    self.send("✅ بستن همه ارسال شد", reply_markup=self._kb())
            else:
                if eng:
                    try:
                        result = eng.close_position(asset)
                        self.send(f"🛑 بستن {asset}: {result}", reply_markup=self._kb())
                    except Exception as e:
                        self.send(f"❌ خطا بستن {asset}: {e}", reply_markup=self._kb())
        else:
            self.send_panel()

    def _bg_loop(self):

    def start_background(self):

            price_str = str(price)[:20]
        try:
            notional_str = f"{float(notional):.2f}"
        except:
            notional_str = str(notional)[:20]
        
        if direction == "buy":
            emoji = "BUY" if intent in ("open","increase") else "CLOSE SHORT"
        else:
            emoji = "SELL" if intent in ("open","increase") else "CLOSE LONG"
        
        lines = [
            f"TRADE {version} - {asset} {emoji}",
            f"Time: {action.get('ts','')}",
            f"Version: {version}",
            f"Instrument: {inst}",
            f"Amount: {amount_str} @ {price_str}",
            f"Notional: ${notional_str}",
            f"Type: {intent} Status: {status}",
            f"Order ID: {order_id}",
        ]
        return "\n".join(lines)
    except Exception as e:
        return f"Trade {action.get('asset','?')} {action.get('direction','?')} {action.get('amount',0)} error:{e}"
